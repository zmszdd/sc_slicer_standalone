contract PriceOracle {
    uint256 public price = 100;

    function getPrice() external view returns (uint256) {
        return price;
    }

    function setPrice(uint256 _price) external {
        price = _price;
    }
}
contract Vault {
    address public owner;
    uint256 public totalAssets;
    uint256 public lastDepositTime;
    PriceOracle public oracle;

    event LogDeposit(address indexed user, uint256 amount);

    constructor(address _oracle) {
        owner = msg.sender;
        oracle = PriceOracle(_oracle);
    }

    // 重点测试函数：包含污点传播与复杂依赖
    function deposit(uint256 _amount) public {
        // 1. 污点源 (Source): _amount 是用户可控的输入
        
        // 2. 控制依赖 (Control Dependency): 逻辑执行取决于条件
        require(_amount > 0, "Amount must be positive");

        // 3. 跨合约依赖 (Cross-contract): 依赖外部 Oracle 的返回值
        uint256 currentPrice = oracle.getPrice();

        // 4. 数据流 (Data Flow): _amount 影响了 valueInUSD
        uint256 valueInUSD = _amount * currentPrice;

        // 5. 干扰项 (Irrelevant for Slicing): 
        // 这一行改变了 lastDepositTime，但与 totalAssets 的最终值无关
        lastDepositTime = block.timestamp;

        // 6. 污点汇聚点 (Sink): 状态变量写入
        totalAssets = totalAssets + valueInUSD;

        // 7. 干扰项: 事件触发对 totalAssets 的数值无计算依赖
        emit LogDeposit(msg.sender, _amount);
    }

    // 用于测试控制依赖下的权限分析
    function withdrawAll() public {
        // 只有 owner 才能影响 totalAssets 清零
        require(msg.sender == owner, "Not owner");
        totalAssets = 0;
    }
}
